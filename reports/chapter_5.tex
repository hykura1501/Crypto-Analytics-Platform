\subsection{Bài học về Kiến trúc Phần mềm}

\begin{enumerate}
    \item \textbf{Design for Failure}:
    \begin{itemize}
        \item Mọi component có thể fail
        \item Cần có retry, circuit breaker, fallback
        \item Graceful degradation thay vì complete failure
    \end{itemize}
    
    \item \textbf{Scalability từ đầu}:
    \begin{itemize}
        \item Stateless services dễ scale
        \item Externalize state vào Redis/Database
        \item Horizontal scaling > Vertical scaling
    \end{itemize}
    
    \item \textbf{Security by Design}:
    \begin{itemize}
        \item Không tin tưởng bất kỳ input nào
        \item Multiple layers of security
        \item Audit logging cho mọi critical operations
    \end{itemize}
    
    \item \textbf{Observability}:
    \begin{itemize}
        \item Logs, Metrics, Traces (ba trụ cột)
        \item Centralized logging (ELK, Loki)
        \item Distributed tracing (Jaeger, Zipkin)
    \end{itemize}
\end{enumerate}
