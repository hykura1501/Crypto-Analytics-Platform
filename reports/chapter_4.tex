\subsection{Database Schema}

\begin{lstlisting}[language=SQL, caption=Main Database Schema]
-- Trading Pairs
CREATE TABLE trading_pairs (
    id SERIAL PRIMARY KEY,
    symbol VARCHAR(20) UNIQUE NOT NULL,
    base_asset VARCHAR(10) NOT NULL,
    quote_asset VARCHAR(10) NOT NULL,
    status VARCHAR(20) DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Price History
CREATE TABLE price_history (
    id SERIAL PRIMARY KEY,
    pair_id INTEGER REFERENCES trading_pairs(id),
    price DECIMAL(20, 8) NOT NULL,
    volume DECIMAL(20, 8) NOT NULL,
    high DECIMAL(20, 8) NOT NULL,
    low DECIMAL(20, 8) NOT NULL,
    open_price DECIMAL(20, 8) NOT NULL,
    close_price DECIMAL(20, 8) NOT NULL,
    timestamp TIMESTAMP NOT NULL,
    interval VARCHAR(10) NOT NULL DEFAULT '1m',
    UNIQUE(pair_id, timestamp, interval)
);

-- News Sources
CREATE TABLE news_sources (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) UNIQUE NOT NULL,
    url VARCHAR(255) NOT NULL,
    selector_type VARCHAR(50) DEFAULT 'css',
    title_selector TEXT,
    content_selector TEXT,
    date_selector TEXT,
    link_selector TEXT,
    status VARCHAR(20) DEFAULT 'active',
    last_crawled_at TIMESTAMP
);

-- News with Sentiment
CREATE TABLE news (
    id SERIAL PRIMARY KEY,
    source_id INTEGER REFERENCES news_sources(id),
    title TEXT NOT NULL,
    content TEXT,
    url VARCHAR(500) UNIQUE NOT NULL,
    author VARCHAR(100),
    published_at TIMESTAMP NOT NULL,
    crawled_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    sentiment_score DECIMAL(5, 4),
    sentiment_label VARCHAR(20),
    keywords TEXT[]
);

-- News Price Alignment
CREATE TABLE news_price_alignment (
    id SERIAL PRIMARY KEY,
    news_id INTEGER REFERENCES news(id) ON DELETE CASCADE,
    pair_id INTEGER REFERENCES trading_pairs(id),
    price_before DECIMAL(20, 8),
    price_after DECIMAL(20, 8),
    price_change_percent DECIMAL(10, 4),
    time_window_hours INTEGER DEFAULT 24,
    correlation_score DECIMAL(5, 4)
);

-- AI Analysis
CREATE TABLE ai_analysis (
    id SERIAL PRIMARY KEY,
    pair_id INTEGER REFERENCES trading_pairs(id),
    analysis_type VARCHAR(50) NOT NULL,
    prediction VARCHAR(20),
    confidence_score DECIMAL(5, 4),
    reasoning TEXT,
    time_horizon VARCHAR(20),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    valid_until TIMESTAMP
);

-- Users
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Watchlists
CREATE TABLE watchlists (
    id SERIAL PRIMARY KEY,
    user_id INTEGER REFERENCES users(id) ON DELETE CASCADE,
    pair_id INTEGER REFERENCES trading_pairs(id) ON DELETE CASCADE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(user_id, pair_id)
);
\end{lstlisting}

\subsection{Trading Chart Component - Hiển thị biểu đồ giá}

Hệ thống sử dụng thư viện \textbf{Lightweight Charts} của TradingView để hiển thị biểu đồ nến (candlestick) với khả năng cập nhật realtime qua WebSocket.

\begin{lstlisting}[language=TypeScript, caption=Chart.tsx - Trading Chart Component (frontend/src/components/Chart.tsx)]
import { createChart, CandlestickSeries, CrosshairMode } from 'lightweight-charts';
import { useWebSocket } from '../contexts/WebSocketContext';
import { apiClient } from '../api/client';

export default function Chart({ symbol = 'BTCUSDT', interval = '1h' }) {
  const chartContainerRef = useRef<HTMLDivElement>(null);
  const chartRef = useRef<IChartApi | null>(null);
  const candlestickSeriesRef = useRef<ISeriesApi<'Candlestick'> | null>(null);
  
  const { subscribe, unsubscribe, lastMessage, getLastMessage, isConnected } =
    useWebSocket();
  
  // Derive current topic message
  const currentTopicMessage = useMemo(() => {
    const topic = `market:${symbol}:${interval}`;
    return getLastMessage(topic);
  }, [symbol, interval, lastMessage, getLastMessage]);

  useEffect(() => {
    // Create chart with professional styling
    const chart = createChart(chartContainerRef.current, {
      layout: {
        background: { color: '#ffffff' },
        textColor: '#333',
      },
      crosshair: {
        mode: CrosshairMode.Normal,
        vertLine: { color: '#9B7DFF', style: 3 },
        horzLine: { color: '#9B7DFF', style: 3 },
      },
      timeScale: { timeVisible: true, secondsVisible: false },
    });

    // Add candlestick series
    const candlestickSeries = chart.addSeries(CandlestickSeries, {
      upColor: '#26a69a',
      downColor: '#ef5350',
      wickUpColor: '#26a69a',
      wickDownColor: '#ef5350',
    });
    
    candlestickSeriesRef.current = candlestickSeries;
    loadHistoricalData(symbol, interval);
    
    return () => chart.remove();
  }, [symbol, interval]);

  const loadHistoricalData = async (sym: string, inter: string) => {
    const response = await apiClient.getHistory({ symbol: sym, interval: inter });
    const formattedData = response.data.map((price) => ({
      time: (new Date(price.time).getTime() / 1000) as Time,
      open: price.open,
      high: price.high,
      low: price.low,
      close: price.close,
    }));
    candlestickSeriesRef.current?.setData(formattedData);
  };
}
\end{lstlisting}

\subsection{Account Management - Quản lý tài khoản}

Hệ thống phân biệt 2 loại tài khoản: \textbf{Free (thường)} và \textbf{VIP (Premium)} với các quyền truy cập khác nhau.

\begin{lstlisting}[language=SQL, caption=User Schema with Role-based Access]
-- Users với account type
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    account_type VARCHAR(20) DEFAULT 'free',  -- 'free' or 'vip'
    vip_expires_at TIMESTAMP,                  -- VIP expiration
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Watchlists cho user quản lý danh sách theo dõi
CREATE TABLE watchlists (
    id SERIAL PRIMARY KEY,
    user_id INTEGER REFERENCES users(id) ON DELETE CASCADE,
    pair_id INTEGER REFERENCES trading_pairs(id) ON DELETE CASCADE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(user_id, pair_id)  -- Mỗi user chỉ có 1 entry cho mỗi pair
);
\end{lstlisting}

\begin{lstlisting}[language=Go, caption=VIP Authorization Middleware]
// VIPRequired middleware checks if user has VIP access
func (h *Handler) VIPRequired() gin.HandlerFunc {
    return func(c *gin.Context) {
        userID := c.GetInt("user_id")
        
        var accountType string
        var vipExpiresAt sql.NullTime
        err := h.DB.QueryRow(
            "SELECT account_type, vip_expires_at FROM users WHERE id = $1",
            userID,
        ).Scan(&accountType, &vipExpiresAt)
        
        if err != nil {
            c.JSON(http.StatusUnauthorized, gin.H{"error": "User not found"})
            c.Abort()
            return
        }
        
        // Check account_type and expiration
        if accountType != "vip" {
            c.JSON(http.StatusForbidden, gin.H{
                "error": "VIP subscription required",
                "upgrade_url": "/upgrade-to-vip",
            })
            c.Abort()
            return
        }
        
        // Check if VIP is still valid
        if vipExpiresAt.Valid && vipExpiresAt.Time.Before(time.Now()) {
            c.JSON(http.StatusForbidden, gin.H{
                "error": "VIP subscription expired",
                "expired_at": vipExpiresAt.Time,
            })
            c.Abort()
            return
        }
        
        c.Next()
    }
}
