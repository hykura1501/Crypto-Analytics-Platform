\section*{Mục tiêu đồ án}
\addcontentsline{toc}{section}{Mục tiêu đồ án}

Đồ án này được thiết kế nhằm phát triển các năng lực cốt lõi về kiến trúc phần mềm cho sinh viên thông qua việc xây dựng một hệ thống phân tích tài chính thời gian thực. Cụ thể, sinh viên sẽ:

\begin{itemize}
    \item \textbf{Nắm vững bản chất kiến trúc hệ thống phân tán}: Hiểu sâu về các thành phần và cơ chế hoạt động của hệ thống tài chính AI, bao gồm xử lý dữ liệu thời gian thực, thu thập dữ liệu quy mô lớn, và đảm bảo bảo mật khi mở rộng.
    
    \item \textbf{Phát triển tư duy phản biện và phân tích kỹ thuật}: Xác định và giải quyết các thách thức kỹ thuật thường gặp trong môi trường production, bao gồm mở rộng kết nối WebSocket cho hàng nghìn người dùng đồng thời, xử lý việc thu thập dữ liệu không ổn định từ nhiều nguồn khác nhau, và bảo vệ hệ thống trước các mối đe dọa bảo mật.
    
    \item \textbf{Thiết kế và đề xuất kiến trúc hợp lý}: Xây dựng các sơ đồ kiến trúc rõ ràng với giải thích chi tiết cho từng thành phần (API Gateway, Load Balancer, Message Broker, Router, Parser, Authentication Service, v.v.).
    
    \item \textbf{So sánh và đánh giá các giải pháp kỹ thuật}: Phân tích trade-offs giữa các phương án thiết kế về các khía cạnh: hiệu năng, chi phí vận hành, khả năng chịu lỗi (fault tolerance), và khả năng mở rộng (scalability).
    
    \item \textbf{Vận dụng AI hỗ trợ phân tích hiệu quả}: Sử dụng Large Language Models như một công cụ hỗ trợ, biết cách đặt câu hỏi (prompt engineering), phản biện và đánh giá kết quả một cách phê phán thay vì chấp nhận máy móc.
    
    \item \textbf{Ứng dụng các pattern và practice hiện đại}: Áp dụng các mẫu thiết kế và thực hành tốt nhất vào các hệ thống thực tế như: dịch vụ tài chính, streaming data, giao dịch thời gian thực, hệ thống phân tích AI, và kiến trúc microservices bảo mật cao.
\end{itemize}

\subsection*{Yêu cầu đối với sinh viên}

Để hoàn thành đồ án này, sinh viên cần thực hiện các nhiệm vụ sau:

\begin{enumerate}
    \item \textbf{Nghiên cứu và phân tích vấn đề}: Tự nghiên cứu các vấn đề kỹ thuật dựa trên tình huống thực tế được mô tả, xác định các yêu cầu và ràng buộc của hệ thống.
    
    \item \textbf{Thiết kế kiến trúc hệ thống}: Xây dựng sơ đồ kiến trúc chi tiết minh họa rõ ràng luồng xử lý dữ liệu, các thành phần và mối quan hệ giữa chúng.
    
    \item \textbf{Đánh giá và lựa chọn giải pháp}: So sánh các phương án khác nhau và giải thích lý do lựa chọn giải pháp cuối cùng dựa trên phân tích trade-offs.
    
    \item \textbf{Sử dụng AI hỗ trợ hiệu quả}: Viết các prompt phù hợp để khai thác Large Language Models, sau đó phản biện và đánh giá kết quả nhận được một cách phê phán.
    
    \item \textbf{Tổng kết và rút kinh nghiệm}: Rút ra các bài học quan trọng về tư duy kiến trúc phần mềm và nâng cao kỹ năng kỹ thuật.
\end{enumerate}

\subsection*{Giới thiệu dự án}

Đồ án tập trung vào việc xây dựng một nền tảng phân tích tài chính dựa trên trí tuệ nhân tạo (\textbf{Crypto Analytics Platform}) với các tính năng chính sau:

\begin{itemize}
    \item \textbf{Hiển thị dữ liệu thời gian thực}: Cung cấp thông tin giá cryptocurrency cập nhật liên tục cho hơn 1,000 người dùng đồng thời thông qua WebSocket connections.
    
    \item \textbf{Thu thập tin tức tự động}: Crawl và tổng hợp tin tức từ nhiều nguồn khác nhau với khả năng tự động thích ứng khi cấu trúc HTML của các trang web thay đổi.
    
    \item \textbf{Phân tích AI tiên tiến}: Áp dụng các mô hình học máy để phân tích sentiment và dự đoán xu hướng giá, kèm theo giải thích chi tiết (Explainable AI).
    
    \item \textbf{Quản lý tài khoản phân cấp}: Hỗ trợ hai loại tài khoản (Free và VIP) với các mức độ truy cập tính năng khác nhau.
    
    \item \textbf{Bảo mật đa lớp}: Đảm bảo an toàn thông tin và phân quyền hợp lý trong kiến trúc microservices phân tán.
\end{itemize}

\subsection*{Yêu cầu chức năng chi tiết}

\begin{mybox}
\textbf{Điều kiện tiên quyết:} Người dùng có thể lựa chọn các cặp tiền điện tử khác nhau để theo dõi (BTCUSDT, ETHUSDT, BNBUSDT, SOLUSDT, ADAUSDT, XRPUSDT, v.v.)
\end{mybox}

\subsubsection*{1. Thu thập tin tức tài chính (News Crawler Service)}
\begin{itemize}
    \item \textbf{Đa nguồn dữ liệu}: Thu thập tin tức tự động từ nhiều trang web tài chính uy tín (CoinTelegraph, CoinDesk, CryptoNews, VNExpress, VnEconomy, v.v.)
    
    \item \textbf{Trích xuất thông tin có cấu trúc}: Xác định và lưu trữ các thông tin quan trọng bao gồm tiêu đề bài viết, nội dung chính, thời gian đăng, tên tác giả, và URL nguồn gốc.
    
    \item \textbf{Học cấu trúc HTML tự động}: Sử dụng AI (Gemini) để tự động phân tích và học cấu trúc HTML của từng trang web, không cần cấu hình thủ công cho mỗi nguồn.
    
    \item \textbf{Thích ứng với thay đổi}: Hệ thống có khả năng tự động điều chỉnh khi các website thay đổi layout hoặc cấu trúc HTML mà không cần can thiệp thủ công.
    
    \item \textbf{Quản lý dữ liệu hiệu quả}: Toàn bộ dữ liệu được lưu trữ đầy đủ trong database, đồng thời chỉ hiển thị thông tin được chọn lọc và định dạng phù hợp trên giao diện người dùng.
\end{itemize}

\subsubsection*{2. Hiển thị biểu đồ giá (Trading Chart)}
\begin{itemize}
    \item \textbf{Giao diện chuyên nghiệp}: Thiết kế tương tự TradingView (\url{https://vn.tradingview.com/chart/}) với các công cụ phân tích kỹ thuật cơ bản.
    
    \item \textbf{Dữ liệu lịch sử}: Tích hợp với Binance REST API để lấy dữ liệu giá lịch sử cho các phân tích và backtest.
    
    \item \textbf{Cập nhật thời gian thực}: Sử dụng Binance WebSocket API để nhận và hiển thị giá mới nhất với độ trễ tối thiểu (< 100ms).
    
    \item \textbf{Đa khung thời gian}: Hỗ trợ nhiều timeframes khác nhau (1 phút, 5 phút, 15 phút, 1 giờ, 4 giờ, 1 ngày, 1 tuần) để phù hợp với nhu cầu phân tích của người dùng.
    
    \item \textbf{Đa cặp giao dịch}: Cho phép theo dõi đồng thời nhiều cặp tiền điện tử (BTCUSDT, ETHUSDT, BNBUSDT, v.v.)
    
    \item \textbf{Kiến trúc có khả năng mở rộng}: Thiết kế để hỗ trợ hàng nghìn người dùng đồng thời với hiệu suất ổn định thông qua WebSocket Hub Pattern.
\end{itemize}

\subsubsection*{3. Phân tích AI và Dự đoán xu hướng (AI Analytics)}
\begin{itemize}
    \item \textbf{Kết hợp đa nguồn dữ liệu}: Align tin tức với dữ liệu giá lịch sử theo time buckets để phân tích mối quan hệ nhân quả giữa sự kiện và biến động giá.
    
    \item \textbf{Mô hình AI đa ngôn ngữ} (sử dụng pre-trained models):
    \begin{itemize}
        \item \textit{FinBERT}: Phân tích sentiment cho văn bản tài chính tiếng Anh với độ chính xác cao trong financial domain.
        \item \textit{PhoBERT}: Phân tích sentiment cho văn bản tiếng Việt, đảm bảo xử lý được tin tức từ các nguồn trong nước.
        \item \textit{XGBoost}: Mô hình dự đoán kết hợp cả technical indicators (RSI, MACD, SMA) và news sentiment scores.
    \end{itemize}
    
    \item \textbf{Explainable AI (XAI) - Trí tuệ nhân tạo có thể giải thích}:
    \begin{itemize}
        \item \textit{Dự đoán xu hướng}: Dự báo hướng di chuyển giá (UP/DOWN) trong các khung thời gian khác nhau (1 giờ, 4 giờ, 24 giờ).
        \item \textit{SHAP Analysis}: Sử dụng SHapley Additive exPlanations để giải thích đóng góp của từng feature vào kết quả dự đoán.
        \item \textit{Top Influential News}: Hiển thị các bài báo có ảnh hưởng lớn nhất kèm theo keywords được trích xuất bằng attention mechanism.
        \item \textit{Confidence Score}: Cung cấp độ tin cậy cho mỗi dự đoán dựa trên feature coverage, news availability và model balance.
    \end{itemize}
\end{itemize}

\subsubsection*{4. Quản lý tài khoản phân cấp (Account Management)}
\begin{itemize}
    \item \textbf{Tài khoản Free (Miễn phí)}:
    \begin{itemize}
        \item Xem biểu đồ giá thời gian thực với các công cụ kỹ thuật cơ bản
        \item Truy cập danh sách tin tức tài chính từ các nguồn đã tổng hợp
        \item Tạo và quản lý watchlist các cặp tiền yêu thích
        \item Tùy chỉnh giao diện và cài đặt hiển thị cá nhân
    \end{itemize}
    
    \item \textbf{Tài khoản VIP (Premium)}:
    \begin{itemize}
        \item Toàn bộ tính năng của tài khoản Free
        \item Truy cập phân tích AI sentiment với điểm số chi tiết
        \item Nhận dự đoán xu hướng giá kèm giải thích bằng SHAP values
        \item Xem mối tương quan (correlation) giữa tin tức và biến động giá
        \item Truy cập Feature Importance Analysis và Top Influential News
        \item API access với rate limit cao hơn
    \end{itemize}
\end{itemize}

\subsection*{Kiến trúc hệ thống tổng quan}

Hệ thống được thiết kế theo kiến trúc microservices với event-driven architecture sử dụng Apache Kafka làm message broker. Kiến trúc này được tổ chức thành các lớp (layers) rõ ràng như sau:

\begin{enumerate}
    \item \textbf{Client Layer (Lớp giao diện người dùng)}: 
    \begin{itemize}
        \item Single-Page Application được xây dựng bằng React, TypeScript và Vite
        \item Tích hợp Lightweight Charts library cho Trading Chart
        \item Các trang chính: Dashboard, News Management, Source Configuration
        \item WebSocket client để nhận dữ liệu real-time
    \end{itemize}
    
    \item \textbf{Microservices Layer (Lớp dịch vụ)}:
    \begin{itemize}
        \item \textit{Auth Service (Golang)}: Xác thực người dùng, quản lý phiên làm việc với JWT + Refresh Token rotation
        \item \textit{Market Service (Golang)}: Tích hợp Binance WebSocket, quản lý price data, triển khai WebSocket Hub Pattern
        \item \textit{Crawler Service (Golang)}: RSS feed parsing, HTML crawling, Kafka producer cho news events
        \item \textit{AI Service (Python)}: FinBERT/PhoBERT sentiment analysis, XGBoost prediction, SHAP explanations
    \end{itemize}
    
    \item \textbf{Message Broker Layer (Lớp truyền thông bất đồng bộ)}: 
    \begin{itemize}
        \item Apache Kafka cluster với các topics: \texttt{news\_new\_article}, \texttt{news\_analyze\_css\_selector}, \texttt{market\_price\_update}
        \item Đảm bảo async communication, decoupling giữa các services
    \end{itemize}
    
    \item \textbf{Data Layer (Lớp dữ liệu)}:
    \begin{itemize}
        \item PostgreSQL 15: Persistent storage cho users, articles, market data, predictions
        \item Redis 7: Caching layer cho frequent queries, session storage, rate limiting
    \end{itemize}
    
    \item \textbf{External Services (Dịch vụ bên ngoài)}:
    \begin{itemize}
        \item Binance API: REST API (lịch sử giá) và WebSocket (real-time)
        \item RSS Feeds: CoinDesk, CoinTelegraph, VNExpress, VnEconomy
        \item Google Gemini AI: Phân tích CSS selectors tự động
    \end{itemize}
\end{enumerate}
